# socket with td_api objects carried as JSON (docs/restructuring.md). The UI keeps the
# same Client surface either way, so a regression is one flag away from being bisected.
#
# Off by default because it needs two things a plain checkout does not have: a running
# meegramd, and the client-direction JSON codec, which TDLib does not ship and
# tools/setup-dependencies.sh generates into the td/ checkout.

    set(TDLIB_SOURCE_DIR "${CMAKE_SOURCE_DIR}/td" CACHE PATH
        "TDLib source checkout left by tools/setup-dependencies.sh. Needed for the JSON
         codec headers, which TDLib does not install.")

    # td/utils/config.h is generated by TDLib's own build, so the build tree is needed
    # too, not just the source checkout.
    set(TDLIB_BUILD_DIR "${CMAKE_SOURCE_DIR}/build/tdlib" CACHE PATH
        "TDLib cross-build tree, for the generated td/utils/config.h")

    set(MEEGRAM_CODEC_DIR "${TDLIB_SOURCE_DIR}/td/generate/auto")

    # Missing generated sources otherwise surface as an include error 200 lines deep.
    if (NOT EXISTS "${MEEGRAM_CODEC_DIR}/td/telegram/td_api_json_client.h")
        message(FATAL_ERROR
            "MEEGRAM_JSON_TRANSPORT needs the client-direction JSON codec, which TDLib does "
            "not ship. Looked in ${MEEGRAM_CODEC_DIR}/td/telegram. Run tools/setup-dependencies.sh "
            "to generate it, or set -DTDLIB_SOURCE_DIR=...")
    endif()

    # Files 1-9 only. File 0 carries the generator's server-direction trailer and nothing
    # else - it does not compile in client mode, and src/JsonCodec.cpp replaces it.
    # tools/generate_json_client.cpp explains why that split is reliable.
    set(MEEGRAM_CODEC_SOURCES "")
    foreach(index RANGE 1 9)
        list(APPEND MEEGRAM_CODEC_SOURCES "${MEEGRAM_CODEC_DIR}/td/telegram/td_api_json_client_${index}.cpp")
    endforeach()

    # -w, because the meegram target builds with -Wall -Wextra -pedantic and, in Debug,
    # -Werror. This is generated TDLib code; its warnings are not ours and -Werror would
    # simply refuse to build. Source-level options are emitted after the target's, so this
    # wins.
    #
    # SKIP_AUTOMOC because AUTOMOC is on for the target and would otherwise scan 1.2 MB of
    # generated C++ for a Q_OBJECT that is definitionally not there.
    set_source_files_properties(${MEEGRAM_CODEC_SOURCES} PROPERTIES COMPILE_OPTIONS "-w" SKIP_AUTOMOC ON)

    # NotificationEndpoint rather than NotificationManager: meegramd posts the
    # notifications under this transport and the app keeps only the tap endpoint. The two
    # are alternatives, not layers - building both would mean two processes racing to post
    # the same banner.
    set(MEEGRAM_CLIENT_SOURCES src/ClientProxy.cpp src/JsonCodec.cpp src/NotificationEndpoint.cpp ${MEEGRAM_CODEC_SOURCES})
    set(MEEGRAM_CLIENT_HEADERS src/NotificationEndpoint.hpp)
else()
    set(MEEGRAM_CLIENT_SOURCES src/Client.cpp src/NotificationManager.cpp)
    set(MEEGRAM_CLIENT_HEADERS src/NotificationManager.hpp)
endif()

    # app - build and run it once after regenerating the codec, or after moving the td/

    add_executable(json_roundtrip EXCLUDE_FROM_ALL tools/json_roundtrip.cpp src/JsonCodec.cpp ${MEEGRAM_CODEC_SOURCES})

    target_compile_options(json_roundtrip PRIVATE -UNDEBUG)

    if (BUILD_HARMATTAN)
        target_compile_options(json_roundtrip PRIVATE
            -march=armv7-a -mtune=cortex-a8 -mfpu=neon -mfloat-abi=hard -mthumb)

            -Wl,-rpath,/opt/meegram/lib -Wl,--dynamic-linker=/lib/ld-linux.so.3 -pie -rdynamic)
        target_link_libraries(json_roundtrip PRIVATE -static-libstdc++)
    endif()

    # The daemon. Deliberately minimal - td_json_client is installed public API, so this
    # links neither td_api nor Qt, and shares no header with the UI beyond the socket
    # path.
    # libdbus, not QtDBus: meegramd claims com.meegram.Daemon so dbus-daemon can activate
    # it, and that is the only D-Bus it speaks. Linking QtCore for one name registration
    # would add megabytes to the process whose small resident set is the entire point.
    # find_library/find_path rather than pkg_check_modules. dbus-1.pc reports its include
    # paths relative to the sysroot ("-I/usr/include/dbus-1.0"), and pkg-config does not
    # prefix them unless PKG_CONFIG_SYSROOT_DIR is set - so the imported target pointed at
    # the *host's* /usr/include/dbus-1.0 and configure failed on a path that does not
    # exist. CMake's own find_* honours CMAKE_SYSROOT and CMAKE_FIND_ROOT_PATH, which
    # tools/toolchain.cmake already sets, and which is how find_package(Td) resolves.
    find_library(DBUS_LIBRARY dbus-1)
    find_path(DBUS_INCLUDE_DIR dbus/dbus.h PATH_SUFFIXES dbus-1.0)

    # dbus-arch-deps.h is generated per architecture and installed beside the library
    # rather than under an include prefix, so it is derived from the library's location
    # instead of searched for - there is no prefix layout that would find it.
    if (DBUS_LIBRARY)
        get_filename_component(DBUS_LIBRARY_DIR "${DBUS_LIBRARY}" DIRECTORY)
        set(DBUS_ARCH_INCLUDE_DIR "${DBUS_LIBRARY_DIR}/dbus-1.0/include")
    endif()

    if (NOT DBUS_LIBRARY OR NOT DBUS_INCLUDE_DIR OR NOT EXISTS "${DBUS_ARCH_INCLUDE_DIR}/dbus/dbus-arch-deps.h")
        message(FATAL_ERROR
            "MEEGRAM_JSON_TRANSPORT needs dbus-1, which meegramd uses to claim "
            "com.meegram.Daemon so dbus-daemon can activate it. Install libdbus-1-dev for "
            "a host build; for Harmattan check the sysroot has usr/include/dbus-1.0 and "
            "usr/lib/dbus-1.0/include.")
    endif()

    add_executable(meegramd src/daemon/main.cpp src/daemon/Notifier.cpp)
    target_link_libraries(meegramd PRIVATE Td::TdJsonStatic Threads::Threads ${DBUS_LIBRARY})
    target_compile_options(meegramd PRIVATE -Wall -Wextra -Wno-psabi)

    # SYSTEM: dbus-protocol.h has pre-C++11 string-literal concatenation that warns under
    # -Wall, and it is not ours to fix. The tdutils headers are the notifier's JSON
    # reader - not installed by TDLib, hence the source and build trees, same as the app.
    # No extra link: Td::TdJsonStatic already pulls Td::tdutils, which is where
    # json_decode lives.
    target_include_directories(meegramd SYSTEM PRIVATE
        ${DBUS_INCLUDE_DIR}
        ${DBUS_ARCH_INCLUDE_DIR}
        ${TDLIB_SOURCE_DIR}/tdutils
        ${TDLIB_BUILD_DIR}/tdutils)

    # TDLib's Clocks uses clock_gettime, which on Harmattan's glibc 2.10 lives in librt
    # rather than libc. The meegram target never trips over this because Qt drags librt
    # in; a Qt-free binary has to ask. Same reasoning as the json_bench target above.
    find_library(RT_LIBRARY rt)
    if (RT_LIBRARY)
        target_link_libraries(meegramd PRIVATE ${RT_LIBRARY})
    endif()

    # The notifier's self-check: JSON in, banner text out, with no phone in the loop. Not
    # wired into the app - build and run it after touching src/daemon/Notifier.cpp, the
    # same way as json_roundtrip. See tools/notifier_check.cpp.
    add_executable(notifier_check EXCLUDE_FROM_ALL tools/notifier_check.cpp src/daemon/Notifier.cpp)
    target_include_directories(notifier_check PRIVATE src/daemon)
    target_include_directories(notifier_check SYSTEM PRIVATE
        ${DBUS_INCLUDE_DIR}
        ${DBUS_ARCH_INCLUDE_DIR}
        ${TDLIB_SOURCE_DIR}/tdutils
        ${TDLIB_BUILD_DIR}/tdutils)
    target_link_libraries(notifier_check PRIVATE Td::TdJsonStatic Threads::Threads ${DBUS_LIBRARY})

    if (RT_LIBRARY)
        target_link_libraries(notifier_check PRIVATE ${RT_LIBRARY})
    endif()

    # The whole file is assertions, and CMAKE_CXX_FLAGS_RELEASE carries -DNDEBUG.
    target_compile_options(notifier_check PRIVATE -UNDEBUG)

    if (BUILD_HARMATTAN)
        # Same flags as json_roundtrip above. Worth having on a check that only ever runs
        # by hand: the interesting place to run this one is the device, where the platform
        # notification manager exists, and without the loader and rpath below a
        # cross-built binary does not start at all.
        target_compile_options(notifier_check PRIVATE
            -march=armv7-a -mtune=cortex-a8 -mfpu=neon -mfloat-abi=hard -mthumb)
        target_link_options(notifier_check PRIVATE
            -Wl,-rpath,/opt/meegram/lib -Wl,--dynamic-linker=/lib/ld-linux.so.3 -pie -rdynamic)
        target_link_libraries(notifier_check PRIVATE -static-libstdc++)
    endif()

    if (BUILD_HARMATTAN)
        # Same ABI and loader flags as the meegram target below, and for the same
        # reasons. Not boostable: meegramd is a daemon, not a launched application.
        target_compile_options(meegramd PRIVATE
            -march=armv7-a -mtune=cortex-a8 -mfpu=neon -mfloat-abi=hard -mthumb)
        target_link_options(meegramd PRIVATE
            -Wl,-rpath,/opt/meegram/lib -Wl,--dynamic-linker=/lib/ld-linux.so.3 -pie -rdynamic)
        target_link_libraries(meegramd PRIVATE -static-libstdc++)

        set_target_properties(meegramd PROPERTIES
            INSTALL_RPATH "/opt/meegram/lib"
            INSTALL_RPATH_USE_LINK_PATH FALSE
            BUILD_WITH_INSTALL_RPATH FALSE)

        install(TARGETS meegramd DESTINATION /opt/meegram/bin)

        # What makes the daemon start on demand: dbus-daemon reads this and execs

        install(FILES resources/com.meegram.Daemon.service DESTINATION /usr/share/dbus-1/services)
    endif()
endif()

if (BUILD_HARMATTAN)
    # Target ABI for the N9's OMAP3630 Cortex-A8. -mthumb shrinks code, which helps
    # the small i-cache.
    #
    # hard, not softfp: the Harmattan sysroot's libc reports
    #   Tag_ABI_VFP_args: VFP registers
    # i.e. floating-point arguments are passed in VFP registers. Building softfp
    # against it fails at link with "uses VFP register arguments, output does not".
    # Re-check with:
    #   readelf -A "$SYSROOT/lib/libc.so.6" | grep Tag_ABI_VFP_args
    target_compile_options(meegram PRIVATE
        -march=armv7-a -mtune=cortex-a8 -mfpu=neon -mfloat-abi=hard -mthumb)

    target_compile_options(meegram PRIVATE ${boostable_CFLAGS})
    target_include_directories(meegram PRIVATE ${boostable_INCLUDE_DIRS})

    target_link_options(meegram PRIVATE -Wl,-rpath,/opt/meegram/lib -Wl,--dynamic-linker=/lib/ld-linux.so.3 -pie -rdynamic)

    target_link_libraries(meegram PRIVATE ${boostable_LIBRARIES} $<$<BOOL:${SANITIZE_ADDRESS}>:-static-libasan> -static-libstdc++)

    # CMake rewrites RPATH on install and, by default, drops everything - the
    # -Wl,-rpath above only applies to the binary in the build tree. Without this the
    # installed binary has no RPATH at all and cannot find the libraries below.
    # BUILD_WITH_INSTALL_RPATH stays off so the build-tree binary keeps the paths
    # that let it link.
    set_target_properties(meegram PROPERTIES
        INSTALL_RPATH "/opt/meegram/lib"
        INSTALL_RPATH_USE_LINK_PATH FALSE
        BUILD_WITH_INSTALL_RPATH FALSE)

    install(FILES resources/meegram_harmattan.desktop DESTINATION /usr/share/applications)
    install(FILES resources/meegram80.png DESTINATION /usr/share/icons/hicolor/80x80/apps)

    # Lets D-Bus start the app when a notification is tapped after it has been closed.
    install(FILES resources/com.meegram.service DESTINATION /usr/share/dbus-1/services)
    install(FILES resources/meegram-splash.png DESTINATION /opt/meegram/share)

    install(TARGETS meegram DESTINATION /opt/meegram/bin)

    # TDLib is linked statically, but it pulls in OpenSSL *shared*, so the binary
    # needs libssl.so.3 / libcrypto.so.3 at runtime. Harmattan ships OpenSSL 0.9.8,
    # so ours has to travel inside the package or the app dies at startup with
    # "libcrypto.so.3: cannot open shared object file".
    #
    # tools/setup-dependencies.sh leaves them in build/crypto/lib rather than
    # installing them into the sysroot; point this elsewhere if that changes.
    set(MEEGRAM_OPENSSL_LIB_DIR "${CMAKE_SOURCE_DIR}/build/crypto/lib"
        CACHE PATH "Directory holding the cross-built libssl/libcrypto to ship")

    # install(DIRECTORY), not file(GLOB) + install(FILES): a glob is evaluated at
    # configure time, so configuring before the dependencies finish building silently
    # bakes in an empty list and the package ships without OpenSSL. This form is
    # evaluated during install, and handles DESTDIR correctly.
    # Versioned sonames only. readelf shows the binary needs libssl.so.3 and
    # libcrypto.so.3; the unversioned libssl.so / libcrypto.so are link-time names.
    # setup-dependencies.sh copies with plain cp, so those are full duplicates rather
    # than symlinks - shipping them would add ~5.7 MB of the same bytes twice.
    install(DIRECTORY "${MEEGRAM_OPENSSL_LIB_DIR}/"
            DESTINATION /opt/meegram/lib
            FILES_MATCHING
                PATTERN "libssl.so.*"
                PATTERN "libcrypto.so.*")

    # dpkg-buildpackage reads debian/ from the build directory, so it has to be copied
    # there - and copied on every run. This used to be a configure-time file(COPY)
    # guarded by "if not exists", which meant the build tree kept the debian/ of its
    # first-ever configure forever: editing debian/changelog had no effect and the
    # package kept coming out with the old version number.
    add_custom_target(package
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/debian ${CMAKE_BINARY_DIR}/debian
